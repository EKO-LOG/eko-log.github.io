// Parite verilerini çekip data/*.json olarak yazar. Çalıştır: node fetch-data.mjs
import { mkdir, writeFile } from 'node:fs/promises';
const DAYS = 1900, UA = { 'User-Agent': 'Mozilla/5.0 (X11; Linux x86_64) AppleWebKit/537.36 Chrome/124 Safari/537.36' };
const YAHOO = { XAU:'GC=F',XAG:'SI=F',XPT:'PL=F',XPD:'PA=F',BRENT:'BZ=F',WTI:'CL=F',NG:'NG=F',TTF:'TTF=F',HO:'HO=F',RB:'RB=F',
  XCU:'HG=F',ALU:'ALI=F',REMX:'REMX',LIT:'LIT',URA:'URA',WHEAT:'ZW=F',CORN:'ZC=F',SOY:'ZS=F',COFFEE:'KC=F',SUGAR:'SB=F',COTTON:'CT=F',COCOA:'CC=F',
  BTC:'BTC-USD',ETH:'ETH-USD',BNB:'BNB-USD',SOL:'SOL-USD',XRP:'XRP-USD',ADA:'ADA-USD',DOGE:'DOGE-USD' };
const FIAT = 'EUR GBP TRY JPY CHF CNY CAD AUD NZD SEK NOK DKK PLN CZK HUF RON ISK BRL MXN ZAR INR KRW SGD HKD ILS IDR MYR PHP THB'.split(' ');
const iso = d => d.toISOString().slice(0, 10), sleep = ms => new Promise(r => setTimeout(r, ms));
async function get(url, tries = 4) {
  let err;
  for (let i = 0; i < tries; i++) {
    try { const r = await fetch(url, { headers: UA }); if (r.ok) return await r.json(); err = new Error('HTTP ' + r.status); }
    catch (e) { err = e; }
    await sleep(1500 * (i + 1));
  }
  throw err;
}
async function yahoo(sym) {
  const t1 = Math.floor(Date.now() / 1e3), t0 = t1 - DAYS * 86400;
  for (const h of ['query1', 'query2']) {
    try {
      const j = await get(`https://${h}.finance.yahoo.com/v8/finance/chart/${encodeURIComponent(sym)}?period1=${t0}&period2=${t1}&interval=1d`, 2);
      const r = j.chart.result[0], q = r.indicators.quote[0].close, o = [];
      r.timestamp.forEach((t, i) => { if (q[i] > 0) o.push([iso(new Date(t * 1e3)), +q[i].toFixed(6)]); });
      if (o.length > 1) return o;
    } catch (e) {}
  }
  throw new Error('yahoo ' + sym);
}
async function fiat() {
  const out = {}, end = new Date();
  for (let y = 0; y < 6; y++) { // 1'er yıllık parçalar
    const e = new Date(end.getTime() - y * 365 * 864e5), s = new Date(e.getTime() - 365 * 864e5);
    const j = await get(`https://api.frankfurter.dev/v1/${iso(s)}..${iso(e)}?base=USD&symbols=${FIAT.join(',')}`);
    Object.assign(out, j.rates);
  }
  return out;
}
await mkdir('data', { recursive: true });
let ok = 0, fail = [];
const stamp = new Date().toISOString();
try { await writeFile('data/fiat.json', JSON.stringify({ u: stamp, r: await fiat() })); ok++; console.log('fiat ok'); }
catch (e) { fail.push('fiat'); console.error('fiat', e.message); }
for (const [code, sym] of Object.entries(YAHOO)) {
  try { await writeFile(`data/${code}.json`, JSON.stringify({ u: stamp, d: await yahoo(sym) })); ok++; console.log(code, 'ok'); }
  catch (e) { fail.push(code); console.error(code, e.message); }
  await sleep(700);
}
console.log(`tamam: ${ok}, hata: ${fail.join(', ') || '-'}`);
if (!ok) process.exit(1);

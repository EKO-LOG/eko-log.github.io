// Cloudflare Worker: Yahoo Finance için kendi küçük aracın.
export default {
  async fetch(req) {
    const u = new URL(req.url);
    if (!u.pathname.startsWith('/v8/finance/chart/')) return new Response('not found', { status: 404 });
    const r = await fetch('https://query1.finance.yahoo.com' + u.pathname + u.search, {
      headers: { 'User-Agent': 'Mozilla/5.0' },
      cf: { cacheTtl: 3600, cacheEverything: true }
    });
    return new Response(r.body, { status: r.status, headers: {
      'Content-Type': 'application/json',
      'Access-Control-Allow-Origin': '*',
      'Cache-Control': 'public, max-age=3600'
    }});
  }
}
